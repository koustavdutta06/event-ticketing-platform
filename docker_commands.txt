stop and remove volume :- docker compose down -v

docker compose down       # stops and removes containers, KEEPS the volume
docker compose up -d      # restarts, reattaches to the same volume, data intact

docker compose stop       # just stops, doesn't remove anything
docker compose start      # resumes the same containers
